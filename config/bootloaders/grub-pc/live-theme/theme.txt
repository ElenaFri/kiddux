# theme.txt — thème graphique GRUB de Kiddux
# =============================================================================
# Menu d'amorçage (UEFI / loopback) aux couleurs du projet, en remplacement de
# l'apparence standard du GRUB Debian (menu texte cyan sur bleu). Fond aubergine
# #2D1535, accents magenta #E07BE8 / #A82E73 : accordé au splash Plymouth
# « kids » et au thème Yaru-magenta du bureau.
#
# Activation : /boot/grub/theme.cfg (fourni par live-build) fait
#   « set theme=/boot/grub/live-theme/theme.txt » DÈS QUE /boot/grub/splash.png
# existe. On dépose donc ce thème ET le splash.png via config/bootloaders/
# grub-pc/ (surcharge canonique de live-build, prise en compte par grub-efi).
# config.cfg charge déjà la police, « insmod png », « insmod gfxterm » puis
# « terminal_output gfxterm » : rien d'autre à activer.
# =============================================================================

# -- Fond -------------------------------------------------------------------
# Image de marque (logo du projet + mot-symbole « debian 13 ») centrée sur un
# aplat aubergine. « padding » : l'image n'est jamais déformée, reste entière et
# centrée, les marges étant comblées par desktop-color — rendu propre à toute
# résolution (l'image est transparente hors de la marque, d'où l'aubergine
# uniforme). desktop-image est relatif à ce fichier : ../splash.png =
# /boot/grub/splash.png.
desktop-image: "../splash.png"
# "crop" : l'image remplit tout l'écran (mise à l'échelle en conservant les
# proportions, débords rognés). On évite "padding", dont GRUB laisse les marges
# en NOIR (il ignore desktop-color là), d'où des bandes noires sur écran large.
# Le splash est en 16:10 (1920x1200) avec de larges marges aubergine, donc le
# rognage ne touche jamais le logo ni le texte.
desktop-image-scale-method: "crop"
desktop-color: "#2d1535"

# Pas de titre texte (la marque est déjà dans l'image).
title-text: ""
message-color: "#f2e8d6"
message-bg-color: "#2d1535"
terminal-font: "Unifont Regular 16"

# -- Aide de navigation (discrète, en bas) ----------------------------------
# @KEYMAP_SHORT@ est remplacé par GRUB au démarrage par un rappel des touches.
+ label {
    top   = 100%-52
    left  = 0
    width = 100%
    height = 20
    text  = "@KEYMAP_SHORT@"
    align = "center"
    color = "#b99fc4"
    font  = "Unifont Regular 16"
}

# -- Menu des entrées (moitié basse de l'écran) -----------------------------
# Sélection signalée par le contraste : blanc vif / lavande. Pas d'icônes.
+ boot_menu {
    left   = 18%
    width  = 64%
    top    = 57%
    height = 30%
    item_color          = "#d9b3e0"
    item_font           = "Unifont Regular 16"
    selected_item_color = "#ffffff"
    selected_item_font  = "Unifont Regular 16"
    item_height  = 24
    item_padding = 4
    item_spacing = 6
    icon_width   = 0
    icon_height  = 0
    item_icon_space = 0
}

# -- Compte à rebours du démarrage automatique ------------------------------
# id = __timeout__ : GRUB y affiche le décompte (texte @TIMEOUT_NOTIFICATION_
# LONG@ traduit). La barre se vide en magenta sur fond aubergine.
+ progress_bar {
    id    = "__timeout__"
    left  = 20%
    top   = 100%-82
    width = 60%
    height = 18
    font  = "Unifont Regular 16"
    text_color   = "#ffffff"
    fg_color     = "#e07be8"
    bg_color     = "#3a1d42"
    border_color = "#a82e73"
    text = "@TIMEOUT_NOTIFICATION_LONG@"
}
